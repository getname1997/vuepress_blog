### 数值类型 

1. int 整数

   ```pyhon
   
   print(int(3.14))     # 3
   print(10 + 5)    # 加法: 15
   
   print(10 - 5)    # 减法: 5
   
   print(10 * 5)    # 乘法: 50
   
   print(10 / 3)    # 除法: 3.333... (结果是浮点数)
   
   print(10 // 3)   # 整除: 3
   
   print(10 % 3)    # 取余: 1
   
   print(2 ** 3)    # 幂运算: 8
   ```

   

2. float 浮点数

   ~~~python
   # 浮点数类型
   pi = 3.14159
   price = 19.99
   temperature = -5.5
   
   # 浮点数运算
   print(0.1 + 0.2)        # 注意浮点数精度问题: 0.30000000000000004
   print(round(0.1 + 0.2, 2))  # 四舍五入: 0.3
   
   # 类型转换
   print(float(10))        # 10.0
   print(float("3.14"))    # 3.14
   
   ~~~

   

3. 复数 complex

       ~~~python
   # 复数类型
   c1 = 3 + 4j
   c2 = 2 - 1j
   
   print(c1 + c2)      # (5+3j)
   print(c1 * c2)      # (10+5j)
   print(c1.real)      # 实部: 3.0
   print(c1.imag)      # 虚部: 4.0
   
       ~~~

   

### 字符串

 1. string 字符串

    ```python
    # 字符串创建
    
    s1 = 'Hello, World!'
    
    s2 = "Python Programming"
    
    s3 = '''多行
    
    字符串'''
    
    s4 = """另一个
    
    多行字符串"""
    
    
    # 字符串索引和切片
    
    text = "Python"
    
    print(text[0])      # 'P' (第一个字符)
    
    print(text[-1])     # 'n' (最后一个字符)
    
    print(text[2:5])    # 'tho' (切片)
    
    print(text[:3])     # 'Pyt' (从头开始)
    
    print(text[3:])     # 'hon' (到末尾)
    
    
    # 字符串常用方法
    
    s = "  Hello World!  "
    
    print(s.strip())        # 去除两端空格: "Hello World!"
    
    print(s.lower())        # 转小写: "  hello world!  "
    
    print(s.upper())        # 转大写: "  HELLO WORLD!  "
    
    print(s.replace("World", "Python"))  # 替换: "  Hello Python!  "
    
    print(s.split())        # 分割: ['Hello', 'World!']
    
    print(len(s))           # 长度: 17
    
    print("Hello" in s)     # 成员检查: True
    
    
    # 字符串格式化
    
    name = "Alice"
    
    age = 25
    
    # f-string (推荐)
    
    print(f"My name is {name} and I'm {age} years old.")
    
    # format方法
    
    print("My name is {} and I'm {} years old.".format(name, age))
    
    # 百分号格式化
    
    print("My name is %s and I'm %d years old." % (name, age))
    ```

### list 数组

  ~~~python
# 列表创建
numbers = [1, 2, 3, 4, 5]
fruits = ["apple", "banana", "cherry"]
mixed = [1, "hello", 3.14, True]
empty_list = []

# 列表索引和切片
print(numbers[0])       # 1

fruits.insert(1, "grape")   # 插入元素: ['apple', 'grape', 'banana', 'cherry', 'orange']
fruits.remove("banana")     # 移除元素: ['apple', 'grape', 'cherry', 'orange']
  ~~~

### 元组 (Tuple)

```python
# 元组创建（不可变）
coordinates = (10, 20)
colors = ("red", "green", "blue")
single_tuple = (5,)  # 注意逗号，表示单元素元组

# 元组操作
print(coordinates[0])       # 10
print(coordinates[1])       # 20
print(colors[:2])           # ('red', 'green')

# 元组解包
x, y = coordinates
print(f"x: {x}, y: {y}")    # x: 10, y: 20

# 元组不可变，但可以重新赋值
coordinates = (30, 40)  # 这是创建新元组，不是修改
```



### 字典 (Dictionary)

```python
# 字典创建
person = {"name": "Alice", "age": 25, "city": "New York"}
empty_dict = {}
grades = dict(math=90, science=85, english=92)

# 访问字典
print(person["name"])       # Alice
print(person.get("age"))    # 25
print(person.get("email", "Not provided"))  # 提供默认值

# 修改字典
person["age"] = 26              # 修改值
person["email"] = "alice@example.com"  # 添加新键值对
person.update({"city": "Boston", "job": "Engineer"})  # 批量更新

# 字典常用方法
keys = person.keys()        # 所有键
values = person.values()    # 所有值
items = person.items()      # 所有键值对
print("name" in person)     # 检查键是否存在: True

# 遍历字典
for key, value in person.items():
    print(f"{key}: {value}")

# 字典推导式
square_dict = {x: x**2 for x in range(5)}  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}
```



###  集合 (Set)

~~~python
# 集合创建（无序、不重复）
fruits_set = {"apple", "banana", "cherry"}
numbers_set = {1, 2, 3, 4, 5}
empty_set = set()  # 注意：不能使用 {} 创建空集合，那会创建字典

# 集合操作
fruits_set.add("orange")        # 添加元素
fruits_set.remove("banana")     # 移除元素，不存在会报错
fruits_set.discard("grape")     # 移除元素，不存在不会报错

# 集合运算
A = {1, 2, 3, 4}
B = {3, 4, 5, 6}

print(A | B)    # 并集: {1, 2, 3, 4, 5, 6}
print(A & B)    # 交集: {3, 4}
print(A - B)    # 差集: {1, 2}
print(A ^ B)    # 对称差集: {1, 2, 5, 6}

# 集合推导式
even_squares = {x**2 for x in range(10) if x**2 % 2 == 0}  # {0, 4, 16, 36, 64}

~~~

### 类型
