## 类型转换 CAST

~~~sql

-- 字符串转整数
SELECT CAST('123' AS INT);          -- 结果: 123
SELECT CAST('123.45' AS INT);       -- 结果: 123 (截断小数)
SELECT CAST('123.45' AS DECIMAL);   -- 结果: 123.45

-- 字符串转小数（指定精度）
SELECT CAST('123.4567' AS DECIMAL(5,2));  -- 结果: 123.46 (四舍五入)
SELECT CAST('123.4567' AS NUMERIC(8,4));  -- 结果: 123.4567

-- 数值转字符串
SELECT CAST(123 AS VARCHAR(10));    -- 结果: '123'
SELECT CAST(123.45 AS VARCHAR(10)); -- 结果: '123.45'
SELECT CAST(123.4567 AS VARCHAR(10)); -- 结果: '123.4567'

-- 字符串转日期
SELECT CAST('2023-12-25' AS DATE);           -- 结果: 2023-12-25
SELECT CAST('2023-12-25 14:30:00' AS DATETIME); -- 结果: 2023-12-25 14:30:00

-- 日期转字符串（某些数据库需要指定格式）
SELECT CAST(GETDATE() AS VARCHAR(20));       -- 结果: '2023-12-25 14:30:00'

-- 提取日期部分
SELECT CAST('2023-12-25 14:30:00' AS DATE);  -- 结果: 2023-12-25

-- 字符串/数值转布尔
SELECT CAST('true' AS BOOLEAN);     -- 结果: true
SELECT CAST('false' AS BOOLEAN);    -- 结果: false
SELECT CAST(1 AS BOOLEAN);          -- 结果: true (某些数据库)
SELECT CAST(0 AS BOOLEAN);          -- 结果: false

-- 布尔转字符串/数值
SELECT CAST(TRUE AS VARCHAR(5));    -- 结果: 'true'
SELECT CAST(FALSE AS INT);          -- 结果: 0


-- 整数类型
CAST(column AS INT)          -- 整数
cast(colum as UNSIGNED)      -- 正整数
CAST(column AS BIGINT)       -- 大整数
CAST(column AS SMALLINT)     -- 小整数
CAST(column AS TINYINT)      -- 微整数

-- 小数类型
CAST(column AS DECIMAL(p,s)) -- 精确小数，p=总位数，s=小数位数
CAST(column AS NUMERIC(p,s)) -- 同DECIMAL
CAST(column AS FLOAT)        -- 浮点数
CAST(column AS REAL)         -- 单精度浮点数


CAST(column AS CHAR(n))      -- 定长字符串
CAST(column AS VARCHAR(n))   -- 变长字符串
CAST(column AS TEXT)         -- 长文本
CAST(column AS NCHAR(n))     -- Unicode定长
CAST(column AS NVARCHAR(n))  -- Unicode变长


CAST(column AS DATE)         -- 日期
CAST(column AS TIME)         -- 时间
CAST(column AS DATETIME)     -- 日期时间
CAST(column AS TIMESTAMP)    -- 时间戳
CAST(column AS YEAR)         -- 年份

CAST(column AS BINARY(n))    -- 二进制
CAST(column AS VARBINARY(n)) -- 变长二进制
CAST(column AS XML)          -- XML数据
CAST(column AS JSON)         -- JSON数据（MySQL 5.7+/PostgreSQL）
CAST(column AS UUID)         -- UUID（PostgreSQL）

~~~

## 字符串替换 REPLACE

~~~sql
-- 标准语法
REPLACE(string, search_string, replacement_string)

-- 参数说明：
-- string: 要进行替换操作的原始字符串
-- search_string: 要查找并替换的子字符串
-- replacement_string: 用来替换的新字符串

~~~



## 数字四舍五入 ROUND

~~~sql
ROUND(numeric_expression, length [, function])

-- 参数说明：
-- numeric_expression: 要舍入的数值表达式（列、常量或计算表达式）
-- length: 舍入的精度（小数位数）。可以为正数、负数或零。
-- function (可选): 指定舍入方式。0或缺省表示四舍五入，非0值（如1）表示截断（直接舍去）。


~~~



## 字符串拼接  CONCAT

~~~sql
-- 标准语法
CONCAT(string1, string2 [, string3, ...])

-- 简写操作符（某些数据库）

string1 + string2              -- SQL Server 默认连接操作符
~~~

### GROUP_CONCAT  查询拼接 SEPARATOR 指定分隔符

```sql
SELECT 
    department_id,
    GROUP_CONCAT(employee_name ORDER BY employee_name SEPARATOR ', ') AS employees
FROM employees
GROUP BY department_id;


SELECT name,GROUP_CONCAT(product_name SEPARATOR  '-'),    SUM(orders.total_amount) AS total_amount
FROM customers
         join orders on customers.id = orders.customer_id
         join order_items on orders.id = order_items.order_id
  where product_name like '%鞋%'
group by name
order by customers.name desc  limit 0, 3;

```

### In 指定集合

```sql
-- 王磊订单打9折 in 分别指出每一个订单
update orders set total_amount = total_amount / 0.9 where customer_id in (select id from customers where name = '王磊');
```
