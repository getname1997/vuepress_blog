## 一对多

使用 同一个外键id 来绑定对应数据 

~~~sql
--  部门表
CREATE TABLE `department` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT 'id',
  `name` varchar(50) NOT NULL COMMENT '部门名称',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=11 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='部门表'

-- 员工表
CREATE TABLE `employee` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT 'ID',
  `name` varchar(50) DEFAULT NULL COMMENT '员工姓名',
  `department_id` int DEFAULT NULL COMMENT '部门id',
  PRIMARY KEY (`id`),
  KEY `employee_department_id_fk` (`department_id`),
  CONSTRAINT `employee_department_id_fk` FOREIGN KEY (`department_id`) REFERENCES `department` (`id`) 
    ON DELETE SET NULL
    ON UPDATE SET NULL
)  CHARSET=utf8mb4 COMMENT='员工表'


-- 员工表 employee 使用 department_id 绑定部门 表 部门表更改了 员工表 department_id改成null
~~~



### 多对多 

通过中间表来关联表

~~~sql
-- 首先 第一张表
CREATE TABLE `article` (
 `id` INT NOT NULL AUTO_INCREMENT,
 `title` VARCHAR(50) NOT NULL,
 `content` TEXT NOT NULL,
 PRIMARY KEY (`id`)
) CHARSET=utf8mb4;
-- 第二张表

CREATE TABLE `tag` (
 `id` INT NOT NULL AUTO_INCREMENT,
 `name` VARCHAR(50) NOT NULL,
 PRIMARY KEY (`id`)
);

-- 中间表
CREATE TABLE `article_tag` (
  `article_id` int NOT NULL,
  `tag_id` int NOT NULL,
  PRIMARY KEY (`article_id`,`tag_id`),
  KEY `article_tag_tag_id_fk` (`tag_id`),
  CONSTRAINT `article_tag_article_id_fk` FOREIGN KEY (`article_id`) REFERENCES `article` (`id`) 
    ON DELETE CASCADE 
    ON UPDATE CASCADE,
  CONSTRAINT `article_tag_tag_id_fk` FOREIGN KEY (`tag_id`) REFERENCES `tag` (`id`) 
    ON DELETE CASCADE 
    ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='文章标签对应表'


~~~
